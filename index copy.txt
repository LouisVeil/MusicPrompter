    function loadFolderSongs(event) {
      if (!event || !event.target || !event.target.files) return;
      var files = Array.prototype.slice.call(event.target.files);
      if (!files.length) return;

      // Sort ONLY the newly selected files alphabetically before reading
      files.sort(function(a, b) {
        return a.name.localeCompare(b.name, undefined, { sensitivity: 'base', numeric: true });
      });

      var loadedBatch = new Array(files.length);
      var completedCount = 0;

      files.forEach(function(file, idx) {
        var reader = new FileReader();
        reader.onload = function(e) {
          loadedBatch[idx] = {
            name: file.name.replace(/\.txt$/i, ""),
            content: e.target.result
          };
          completedCount++;

          // Once every file in this batch finishes reading, append in exact A-Z order
          if (completedCount === files.length) {
            loadedBatch.forEach(function(item) {
              loadedSongsMap[item.name] = item.content;
              songSourcesMap[item.name] = "txt";

              if (!(item.name in modifiedSongsStatus)) {
                modifiedSongsStatus[item.name] = false;
              }
              if (playlistOrder.indexOf(item.name) === -1) {
                playlistOrder.push(item.name);
              }
            });

            renderPlaylist();
            saveAppStateToCache();
          }
        };
        reader.readAsText(file);
      });

      event.target.value = "";
    }

    function moveSong(index, direction) {
      var targetIndex = index + direction;
      if (targetIndex < 0 || targetIndex >= playlistOrder.length) return;
      var temp = playlistOrder[index];
      playlistOrder[index] = playlistOrder[targetIndex];
      playlistOrder[targetIndex] = temp;
      renderPlaylist();
    }

    function removeSong(index, songName) {
      playlistOrder.splice(index, 1);
      delete modifiedSongsStatus[songName];
      delete songSourcesMap[songName];
      if (activeSongName === songName) {
        if (isScrolling) toggleScroll();
        activeSongName = null;
        document.getElementById('prompter-stage').innerHTML = 
          "<div id='welcome-message'><h2>Ready to Jam?</h2><p>Select a song from your playlist sidebar to load it onto the stage view.</p></div>";
      }
      renderPlaylist();
    }

    function duplicateSong(index, songName) {
      var baseName = "Copy of " + songName;
      var newName = baseName;
      var counter = 2;

      // Ensure a unique name if "Copy of <songName>" already exists
      while (newName in loadedSongsMap) {
        newName = baseName + " (" + counter + ")";
        counter++;
      }

      loadedSongsMap[newName] = loadedSongsMap[songName] || "";
      songSourcesMap[newName] = songSourcesMap[songName] || "txt";
      modifiedSongsStatus[newName] = true;

      // Insert the duplicate directly below the original song in the playlist
      playlistOrder.splice(index + 1, 0, newName);

      renderPlaylist();
      saveAppStateToCache();
    }

    function selectSong(songName) {
      if (isScrolling) toggleScroll();
      activeSongName = songName;
      renderPlaylist();

      var rawMarkupText = loadedSongsMap[songName];
      var rawLines = rawMarkupText.split(/\r?\n/).map(function(ln) { return ln.trim(); });

      // 1. Render song markup onto the stage and align Left-of-Lyrics columns
      document.getElementById('prompter-stage').innerHTML = compileMarkupToHTML(rawLines);
      syncLeftChordsColumnWidth();
      document.getElementById('viewport').scrollTo(0, 0);

      // 2. Set Metronome BPM first (so Auto Scroll speed uses the new song's tempo)
      var foundTempo = 90;
      var firstMatchIdx = 0;

      rawLines.forEach(function(ln) {
        if (ln.toLowerCase().startsWith("{tempo:")) {
          var rawValue = ln.replace(/\{tempo:\s*/i, "").replace(/\}/, "").trim();
          var digitMatch = rawValue.match(/\d+/);
          if (digitMatch) {
            var extractedTempo = parseInt(digitMatch[firstMatchIdx], 10);
            if (!isNaN(extractedTempo)) foundTempo = extractedTempo;
          }
        }
      });
      document.getElementById('metro-bpm').value = foundTempo;

      // 3. Set Scroll Speed (Auto-calculated + stats banner if checked, or manual {ScrollSpeed:})
      if (document.getElementById('auto-speed-toggle').checked) {
        toggleAutoSpeedState();
      } else {
        document.getElementById('scroll-speed').value = getManualSongScrollSpeed(rawMarkupText);
      }
    }



    // SMOOTH ACCUMULATIVE FRACTIONAL SCROLL ENGINE
    function toggleScroll() {
      var btn = document.getElementById('play-btn');
      var viewport = document.getElementById('viewport');
      
      if (isScrolling) {
        clearInterval(scrollInterval);
        btn.innerText = "▶ Play";
        btn.className = "btn play";
        isScrolling = false;
      } else {
        var speedInput = document.getElementById('scroll-speed');
        var speed = parseFloat(speedInput.value) || 1;
        
        // High-precision background accumulator coordinates
        var exactPositionTracker = viewport.scrollTop;
        
        scrollInterval = setInterval(function() {
          // 1. Add the raw fractional speed to our precise float counter
          exactPositionTracker += speed;
          
          // 2. Extract only the whole completed integer pixels
          var wholePixelsToMove = Math.floor(exactPositionTracker) - viewport.scrollTop;
          
          if (wholePixelsToMove >= 1) {
            viewport.scrollBy(0, wholePixelsToMove);
          }
        }, 16.67); // Stable 60 Frames Per Second calculation grid
        
        btn.innerText = "❚❚ Pause";
        btn.className = "btn stop";
        isScrolling = true;
      }
    }

    function analyzeSongBeatsAndLines(rawText) {
      var lines = rawText.split(/\r?\n/).map(function(ln) { return ln.trim(); });

      var chordIdx = 1;
      var beatIdx = 2;
      var numIdx = 0;
      var denIdx = 1;

      // 1. Detect {Beat: num/den} time signature (defaults to 4 beats per measure)
      var beatsPerMeasure = 4;
      for (var b = 0; b < lines.length; b++) {
        if (lines[b].toLowerCase().startsWith("{beat:")) {
          var beatVal = lines[b].replace(/\{beat:\s*/i, "").replace(/\}/, "").trim();
          var slashParts = beatVal.split("/");
          var topNum = parseFloat(slashParts[numIdx]);
          if (!isNaN(topNum) && topNum > 0) {
            beatsPerMeasure = topNum;
          }
          break;
        }
      }

      // 2. Merge {Merge} lines just like compileMarkupToHTML
      var mergedLines = [];
      for (var i = 0; i < lines.length; i++) {
        var cur = lines[i];
        if (cur.toLowerCase().startsWith("{merge}") && mergedLines.length > 0) {
          mergedLines[mergedLines.length - 1] += " " + cur.substring(7).trim();
          continue;
        }
        mergedLines.push(cur);
      }

      var totalChordBeats = 0;
      var totalContentLines = 0;
      var lastLineBeats = 0;

      mergedLines.forEach(function(ln) {
        if (!ln) return;

        // Check for standalone {Repeat: n} directive on its own line
        var standaloneRepeatMatch = ln.match(/^\{repeat\s*:\s*(\d+)\s*\}$/i);
        if (standaloneRepeatMatch) {
          var repCount = parseInt(standaloneRepeatMatch[chordIdx], 10) || 1;
          if (repCount > 1 && lastLineBeats > 0) {
            totalChordBeats += lastLineBeats * (repCount - 1);
            totalContentLines += (repCount - 1);
          }
          return;
        }

        // Skip all other directive lines like {Title:}, {Para:}, {Empty}, {Tempo:}, etc.
        if (ln.startsWith("{") && ln.endsWith("}")) return;

        // Check for inline {Repeat: n} or legacy [x2] anywhere on a lyric/chord line
        var lineMultiplier = 1;
        var inlineRepeatMatch = ln.match(/\{repeat\s*:\s*(\d+)\s*\}/i) || ln.match(/\[x(\d+)\]/i);
        if (inlineRepeatMatch) {
          lineMultiplier = parseInt(inlineRepeatMatch[chordIdx], 10) || 1;
        }

        // Scan every [Chord] and optional {n} or {1/2}
        var reg = /\[([^\]]+)\](?:\s*\{([^}:]*)\})?/g;
        var m;
        var lineMeasureUnits = 0;

        while ((m = reg.exec(ln)) !== null) {
          var chordLabel = m[chordIdx].trim();
          // Skip legacy [x2] repeat tags
          if (/^x\d+$/i.test(chordLabel)) continue;

          var beatStr = (m[beatIdx] || "").trim();
          var measureFactor = 1; // Default: 1 chord = 1 full measure

          if (beatStr.indexOf("/") !== -1) {
            var parts = beatStr.split("/");
            var num = parseFloat(parts[numIdx]);
            var den = parseFloat(parts[denIdx]);
            if (!isNaN(num) && !isNaN(den) && den !== 0) {
              measureFactor = num / den;
            }
          } else if (beatStr !== "") {
            var parsed = parseFloat(beatStr);
            if (!isNaN(parsed)) {
              measureFactor = parsed;
            }
          }

          lineMeasureUnits += measureFactor;
        }

        var singlePassBeats = lineMeasureUnits * beatsPerMeasure;
        totalChordBeats += singlePassBeats * lineMultiplier;
        totalContentLines += lineMultiplier;
        lastLineBeats = singlePassBeats;
      });

      var beatsPerLine = totalContentLines > 0
        ? (totalChordBeats / totalContentLines).toFixed(2)
        : "0.00";

      return {
        beats: totalChordBeats,
        lines: totalContentLines,
        beatsPerLine: beatsPerLine,
        beatsPerMeasure: beatsPerMeasure
      };
    }

    function getManualSongScrollSpeed(rawText) {
      var foundScrollSpeed = 0.2;
      var firstMatchIdx = 0;
      var lines = rawText.split(/\r?\n/).map(function(ln) { return ln.trim(); });

      lines.forEach(function(ln) {
        if (ln.toLowerCase().startsWith("{scrollspeed:")) {
          var rawSpeedVal = ln.replace(/\{scrollspeed:\s*/i, "").replace(/\}/, "").trim();
          var floatMatch = rawSpeedVal.match(/[\d.]+/);
          if (floatMatch) {
            var extractedSpeed = parseFloat(floatMatch[firstMatchIdx]);
            if (!isNaN(extractedSpeed)) foundScrollSpeed = extractedSpeed;
          }
        }
      });

      return foundScrollSpeed;
    }

    function calculateAutoScrollSpeed(stats) {
      var bpm = parseFloat(document.getElementById('metro-bpm').value) || 90;
      if (stats.beats <= 0 || bpm <= 0) return 0.2;

      // 1. Total song duration in seconds
      var totalSeconds = (stats.beats / bpm) * 60;

      // 2. Actual rendered pixel distance (accounts for font size, chord mode, wrapping, and spacing)
      var stage = document.getElementById('prompter-stage');
      var viewport = document.getElementById('viewport');
      var stageHeight = stage ? stage.scrollHeight : 0;
      var visibleBuffer = viewport ? (viewport.clientHeight * 0.45) : 0;
      var scrollablePixels = Math.max(stageHeight - visibleBuffer, stageHeight * 0.5);

      // 3. Convert pixels/second into pixels/frame at 60 FPS
      var pixelsPerFrame = scrollablePixels / (totalSeconds * 60);

      // Clamp to a safe minimum and round to 2 decimal places
      return Math.max(0.05, Math.round(pixelsPerFrame * 100) / 100);
    }

    function syncLeftChordsColumnWidth() {
      var chordCols = document.querySelectorAll('.left-chords-col');
      if (chordCols.length === 0) return;

      // 1. Reset widths so each cell reports its true natural content width
      chordCols.forEach(function(col) {
        col.style.width = 'auto';
      });

      // 2. Find the widest chord cell across ALL paragraphs in the song
      var maxWidth = 0;
      chordCols.forEach(function(col) {
        var w = col.getBoundingClientRect().width;
        if (w > maxWidth) maxWidth = w;
      });

      // 3. Apply that unified width to every chord cell so all lyrics align vertically
      if (maxWidth > 0) {
        var unifiedWidth = Math.ceil(maxWidth) + 'px';
        chordCols.forEach(function(col) {
          col.style.width = unifiedWidth;
        });
      }
    }

    function parseLyricsLine(line, lSize, cSize, fontName) {
      var chordsLocEl = document.getElementById('opt-chords-location');
      var chordsLocation = chordsLocEl ? chordsLocEl.value : "in-lyrics";
      var showBeatCountEl = document.getElementById('opt-show-beat-count');
      var showBeatCount = showBeatCountEl ? showBeatCountEl.checked : false;

      var chordIdx = 1;
      var beatIdx = 2;
      var repeatIdx = 3;

      function buildChordHtml(chordText, beatNum) {
        var html = chordText;
        if (showBeatCount && beatNum && beatNum.length > 0) {
          html += "<span class='beat-count'>(" + beatNum + ")</span>";
        }
        return html;
      }

      // Matches either [Chord]{optionalBeat} OR {Repeat: n} anywhere on the line
      var tokenReg = /\[([^\]]+)\](?:\s*\{([^}:]*)\})?|\{repeat\s*:\s*(\d+)\s*\}/gi;
      var strippedLine = line.replace(tokenReg, '');
      var hasTxt = strippedLine.trim().length > 0;

      // Chord-only lines (e.g., Intro / Instrumental / Solo)
      if (!hasTxt) {
        var chordOnlyHtml = line.replace(tokenReg, function(_, ch, beat, rep) {
          if (rep) {
            return "<span class='repeat-badge'>(x" + rep + ")</span>";
          }
          return buildChordHtml(ch, beat);
        });
        // Preserve spacing outside HTML tags
        chordOnlyHtml = chordOnlyHtml.replace(/ (?![^<]*>)/g, '&nbsp;');

        return "<div style='font-family:\"" + fontName + "\",monospace; font-size:" + (cSize + globalSizeModifier) + "pt; font-weight:bold; color:var(--chord-color);'>" + chordOnlyHtml + "</div>";
      }

      var m, idx = 0, lLine = "";
      var events = [];

      while ((m = tokenReg.exec(line)) !== null) {
        lLine += line.substring(idx, m.index);

        if (m[repeatIdx]) {
          // {Repeat: n} token found anywhere on the line
          events.push({
            type: "repeat",
            pos: lLine.length,
            repeat: m[repeatIdx]
          });
        } else {
          // [Chord]{optionalBeat} token
          events.push({
            type: "chord",
            pos: lLine.length,
            chord: m[chordIdx],
            beat: m[beatIdx] || ""
          });
        }

        idx = tokenReg.lastIndex;
      }
      lLine += line.substring(idx);

      var fixedLyricsHtml = lLine;

      // MODE 3: None (hide lyric chords, beat counts, and repeat badges)
      if (chordsLocation === "none") {
        return "<div class='line-container'>" +
               "<div class='lyrics-row' style='font-family:\"" + fontName + "\",monospace; font-size:" + (lSize + globalSizeModifier) + "pt; min-height:" + (lSize * 1.2) + "px;'>" + fixedLyricsHtml + "</div></div>";
      }

      // MODE 2: Left of Lyrics (Column 2 = chords + badges, Column 3 = lyrics)
      if (chordsLocation === "left-of-lyrics") {
        var leftParts = [];
        events.forEach(function(ev) {
          if (ev.type === "repeat") {
            var repBadge = "<span class='repeat-badge'>(x" + ev.repeat + ")</span>";
            if (leftParts.length > 0) {
              leftParts[leftParts.length - 1] += repBadge;
            } else {
              leftParts.push(repBadge);
            }
          } else {
            leftParts.push(buildChordHtml(ev.chord, ev.beat));
          }
        });

        return "<div class='left-chords-row'>" +
               "<div class='left-chords-col' style='font-family:\"" + fontName + "\",monospace; font-size:" + (cSize + globalSizeModifier) + "pt;'>" + leftParts.join("&nbsp;&nbsp;&nbsp;") + "</div>" +
               "<div class='left-lyrics-col' style='font-family:\"" + fontName + "\",monospace; font-size:" + (lSize + globalSizeModifier) + "pt;'>" + fixedLyricsHtml + "</div>" +
               "</div>";
      }

      // MODE 1: In Lyrics (stacked chords above lyrics with "(x2)" placed at its exact character column)
      var fixedChordHtml = "";
      var cursorPos = 0;

      events.forEach(function(ev) {
        while (cursorPos < ev.pos) {
          fixedChordHtml += "&nbsp;";
          cursorPos++;
        }

        if (ev.type === "repeat") {
          var badgePlain = "(x" + ev.repeat + ")";
          fixedChordHtml += "<span style='font-size:" + (cSize + globalSizeModifier) + "pt;' class='repeat-badge'>" + badgePlain + "</span>";
          cursorPos += badgePlain.length;
        } else {
          var labelPlain = (showBeatCount && ev.beat.length > 0)
            ? (ev.chord + "(" + ev.beat + ")")
            : ev.chord;

          fixedChordHtml += "<span style='font-size:" + (cSize + globalSizeModifier) + "pt; font-weight:bold; color:var(--chord-color);'>" +
                            buildChordHtml(ev.chord, ev.beat) +
                            "</span>";
          cursorPos += labelPlain.length;
        }
      });

      return "<div class='line-container'>" +
             "<div class='chord-row' style='font-family:\"" + fontName + "\",monospace; font-size:" + (lSize + globalSizeModifier) + "pt; min-height:" + (cSize * 1.2) + "px;'>" + fixedChordHtml + "</div>" +
             "<div class='lyrics-row' style='font-family:\"" + fontName + "\",monospace; font-size:" + (lSize + globalSizeModifier) + "pt; min-height:" + (lSize * 1.2) + "px;'>" + fixedLyricsHtml + "</div></div>";
    }

    function compileMarkupToHTML(lines) {
      var showSongInfo = document.getElementById('opt-show-song-info') ? document.getElementById('opt-show-song-info').checked : false;
      var showStrumInfo = document.getElementById('opt-show-strum-info') ? document.getElementById('opt-show-strum-info').checked : true;
      var showParaName = document.getElementById('opt-show-para-name') ? document.getElementById('opt-show-para-name').checked : true;
      var chordsLocEl = document.getElementById('opt-chords-location');
      var isLeftChords = chordsLocEl && chordsLocEl.value === "left-of-lyrics";
      var cellClass = isLeftChords ? "content-cell left-chords-mode" : "content-cell";



      var lSz = 18, cSz = 14, lp = 12, rp = 88; 
      var fontFm = "Roboto Mono"; 
      var isTab = false, seenFirstPara = false, html = "", comp = [];
      currentSongHeaderInfo = [];
      currentSongLayoutInfo = []

      for (var i = 0; i < lines.length; i++) {
        var cur = lines[i];
        if (cur.toLowerCase().startsWith("{merge}") && comp.length > 0) {
          var textToAppend = cur.substring(7).trim(); 
          comp[comp.length - 1] = comp[comp.length - 1] + " " + textToAppend;
          continue;
        }
        comp.push(cur);
      }

      comp.forEach(function(ln) {
        if (ln.startsWith("{") && ln.endsWith("}")) {
          var d = ln.substring(1, ln.length - 1).trim();
          var cIdx = d.indexOf(":");
          var k = cIdx !== -1 ? d.substring(0, cIdx).trim().toLowerCase() : d.toLowerCase();
          var v = cIdx !== -1 ? d.substring(cIdx + 1).trim() : "";

          if (k === "para") {
            seenFirstPara = true;
            if (isTab) { html += "</td></tr>"; } else { html += "<table class='song-table'>"; isTab = true; }
            if (showParaName) {
              html += "<tr><td class='para-cell'>" + v + "</td><td class='" + cellClass + "'>";
            } else {
              html += "<tr><td class='" + cellClass + "'>";
            }
            return;
          }

          // Collect all text before the first {para} for the Info popup (excluding layout/font directives)
          // Collect all text before the first {para} for Info & Settings popups
          if (!seenFirstPara && ln.trim() !== "") {
            var displayLayoutKeys = ["lyricssize", "chordssize", "lyricschordsfont", "parasplit", "chorddef", "chordpos"];
            var skipKeys = ["empty", "split"];

            if (displayLayoutKeys.indexOf(k) !== -1) {
              currentSongLayoutInfo.push(d);
            } else if (skipKeys.indexOf(k) === -1) {
              currentSongHeaderInfo.push(d);
            }
          }

          if (k === "title") {
            html += "<div class='song-title'>" + v + "</div>";
          } else if (k === "subtitle") {
            html += "<div class='song-subtitle'>" + v + "</div>";
          } else if (k === "lyricssize") {
            lSz = parseFloat(v) || 18;
          } else if (k === "chordssize") {
            cSz = parseFloat(v) || 14;
          } else if (k === "lyricschordsfont") {
            fontFm = v || "Roboto Mono"; 
          } else if (k === "parasplit") {
            var sp = v.split(/\s+/).map(Number);
            if (sp.length === 2 && !isNaN(sp[0]) && !isNaN(sp[1])) { lp = sp[0]; rp = sp[1]; } 
          } else if (k === "empty") {
            html += "<div class='empty-space' style='height:" + (lSz * 1.5) + "px;'></div>";
          } else if (k === "split") {
            html += "<div style='clear:both;'></div>";
          } else if (k !== "chorddef" && k !== "chordpos") {
            var isStrumTag = (k.indexOf("strum") !== -1 || k.indexOf("pick") !== -1);
            if (isStrumTag && !showStrumInfo) return;
            if (!seenFirstPara && !isStrumTag && !showSongInfo) return;

            html += "<div class='meta-line'><b>" + (k.charAt(0).toUpperCase() + k.slice(1)) + ":</b> " + v + "</div>";
          }
        } else {
          if (!seenFirstPara && ln.trim() !== "") {
            currentSongHeaderInfo.push(ln.trim());
            if (!showSongInfo) return;
          }
          var hLn = parseLyricsLine(ln, lSz, cSz, fontFm);
          html += isTab ? hLn : "<div>" + hLn + "</div>";
        }
      });

      if (isTab) html += "</td></tr></table>";
      return html;
    }

    window.addEventListener('keydown', function(event) {
      var activeTag = document.activeElement.tagName.toLowerCase();
      if (activeTag === 'input' || activeTag === 'textarea') return; 

      var speedInput = document.getElementById('scroll-speed');
      var currentSpeed = parseFloat(speedInput.value) || 1;
      var viewport = document.getElementById('viewport');

      switch(event.code) {
        case 'Space':
          event.preventDefault();
          toggleScroll(); 
          break;
          
        case 'ArrowUp':
          event.preventDefault();
          var newSpeedUp = (currentSpeed + 0.2).toFixed(1);
          speedInput.value = newSpeedUp;
          if (isScrolling) {
            clearInterval(scrollInterval);
            scrollInterval = setInterval(function() { viewport.scrollBy(0, parseFloat(newSpeedUp)); }, 16.67);
          }
          break;
          
        case 'ArrowDown':
          event.preventDefault();
          var newSpeedDown = Math.max(0.1, currentSpeed - 0.2).toFixed(1);
          speedInput.value = newSpeedDown;
          if (isScrolling) {
            clearInterval(scrollInterval);
            scrollInterval = setInterval(function() { viewport.scrollBy(0, parseFloat(newSpeedDown)); }, 16.67);
          }
          break;

        case 'ArrowRight':
          event.preventDefault();
          viewport.scrollBy(0, 150); 
          break;

        case 'ArrowLeft':
          event.preventDefault();
          viewport.scrollBy(0, -150); 
          break;
      }
    });

    function toggleMetronome() {
      var btn = document.getElementById('metro-btn');

      }

    function updateMetronomeTempo() {
      if (isMetroRunning) {
        // If running, cycle the background interval immediately to register the new speed
        clearInterval(metroInterval);
        runMetronomeTickerLoop();
      }
    }

    function adjustBPM(amount) {
      var bpmInput = document.getElementById('metro-bpm');
      var currentBPM = parseInt(bpmInput.value) || 120;
      var newBPM = Math.min(300, Math.max(30, currentBPM + amount));
      bpmInput.value = newBPM;
      updateMetronomeTempo();
    }

    function runMetronomeTickerLoop() {
      var bpm = parseInt(document.getElementById('metro-bpm').value) || 120;
      // Formula to convert Beats Per Minute into exact millisecond frame delays
      var intervalMs = (60 / bpm) * 1000;

      // Fire the first beat immediately on start click
      executeSingleMetronomeBeat();

      metroInterval = setInterval(function() {
        executeSingleMetronomeBeat();
      }, intervalMs);
    }

    function executeSingleMetronomeBeat() {
      var visualBox = document.getElementById('metro-visual');
      var soundEnabled = document.getElementById('metro-audio-toggle').checked;
      
      metroBeatCounter++;
      // Standard 4/4 time signature tracking setup loop
      if (metroBeatCounter > 4) metroBeatCounter = 1;

      // 1. VISUAL PULSE ACCENT PROFILES
      if (metroBeatCounter === 1) {
        visualBox.style.background = "#ff3333"; // Bright Red flash on Downbeat (Beat 1)
      } else {
        visualBox.style.background = "#ffb74d"; // Amber flash on weak beats (Beats 2, 3, 4)
      }

      // Quick decay reset animation handler to turn the light off after 100ms
      setTimeout(function() {
        if (isMetroRunning) visualBox.style.background = "#222";
      }, 100);

      // 2. AUDIO SYNTHESIS CLICK CONTROLLER (Web Audio API Node generation)
      if (soundEnabled) {
        try {
          if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
          
          var osc = audioCtx.createOscillator();
          var gain = audioCtx.createGain();
          
          osc.connect(gain);
          gain.connect(audioCtx.destination);
          
          // Accentuate the audio sound pitch on Beat 1 to provide acoustic tracking depth
          if (metroBeatCounter === 1) {
            osc.frequency.setValueAtTime(1000, audioCtx.currentTime); // High pitch click
          } else {
            osc.frequency.setValueAtTime(600, audioCtx.currentTime);  // Low pitch click
          }
          
          // Generate a sudden envelope click strike with zero signal decay hum tail
          gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.05);
          
          osc.start();
          osc.stop(audioCtx.currentTime + 0.06);
        } catch (e) {
          console.log("Audio Engine context delayed or blocked: " + e);
        }
      }
    }

    async function exportPlaylistFile() {
      if (playlistOrder.length === 0) {
        alert("Your playlist is empty. Add some songs before saving a set.");
        return;
      }

      var setlistPackage = {
        playlistOrder: playlistOrder,
        loadedSongsMap: loadedSongsMap,
        modifiedSongsStatus: {}
      };

      var jsonString = JSON.stringify(setlistPackage, null, 2);
      var suggestedFileName = activeSetlistHandleName || "my-setlist.json";

      try {
        if (window.showSaveFilePicker) {
          var options = {
            suggestedName: suggestedFileName,
            types: [{
              description: 'JSON Setlist File',
              accept: { 'application/json': ['.json'] }
            }]
          };

          var fileHandle = await window.showSaveFilePicker(options);
          var writableStream = await fileHandle.createWritable();
          await writableStream.write(jsonString);
          await writableStream.close();

          var savedFileName = fileHandle.name;
          activeSetlistHandleName = savedFileName;

          // Update every song's origin to this new .json setlist and clear dirty flags
          playlistOrder.forEach(function(songName) {
            songSourcesMap[songName] = savedFileName;
            modifiedSongsStatus[songName] = false;
          });

          renderPlaylist();
          saveAppStateToCache();
          alert("Setlist '" + savedFileName + "' saved! All songs in the current list are now linked to this set.");
        } else {
          var blob = new Blob([jsonString], { type: "application/json" });
          var url = URL.createObjectURL(blob);
          var a = document.createElement("a");
          a.href = url;
          a.download = suggestedFileName;
          a.click();
          URL.revokeObjectURL(url);
        }
      } catch (err) {
        if (err.name !== 'AbortError') {
          alert("Error saving setlist file: " + err.toString());
        }
      }
    }

    function importPlaylistFile(event) {
      if (!event || !event.target || !event.target.files) return;
      var files = event.target.files;
      if (!files.length) return;

      var file = files[0];
      var reader = new FileReader();
      reader.onload = function(e) {
        try {
          var parsedPackage = JSON.parse(e.target.result);
          
          if (parsedPackage.playlistOrder && parsedPackage.loadedSongsMap) {
            var importedStatus = parsedPackage.modifiedSongsStatus || {};

            parsedPackage.playlistOrder.forEach(function(songName) {
              loadedSongsMap[songName] = parsedPackage.loadedSongsMap[songName];
              songSourcesMap[songName] = file.name; // Tracks the specific .json set origin
              
              if (!(songName in modifiedSongsStatus)) {
                modifiedSongsStatus[songName] = importedStatus[songName] || false;
              }
              if (playlistOrder.indexOf(songName) === -1) {
                playlistOrder.push(songName);
              }
            });
            
            activeSetlistHandleName = file.name;
            if (!activeSongName && playlistOrder.length > 0) {
              activeSongName = playlistOrder[0];
              selectSong(activeSongName);
            }
            
            renderPlaylist();
            saveAppStateToCache();
            alert("Setlist '" + file.name + "' loaded and added to your list!");
          } else {
            alert("Error: Missing structural configurations inside this setlist file format.");
          }
        } catch (err) {
          alert("Error decoding file contents: " + err.toString());
        }
      };
      reader.readAsText(file);
      event.target.value = ""; 
    }

    function runBatchScript(isPreviewOnly) {
      var code = document.getElementById('batch-script-input').value.trim();
      var statusEl = document.getElementById('batch-preview-status');

      if (!code) {
        statusEl.style.color = "#ff5252";
        statusEl.innerText = "Please enter a script first.";
        return;
      }

      var transformFn;
      try {
        // Creates a sandboxed function receiving (text, name)
        transformFn = new Function("text", "name", code);
      } catch (syntaxErr) {
        statusEl.style.color = "#ff5252";
        statusEl.innerText = "Script Syntax Error: " + syntaxErr.message;
        return;
      }

      var modifiedCount = 0;
      var updatedDrafts = {};

      try {
        for (var i = 0; i < playlistOrder.length; i++) {
          var songName = playlistOrder[i];
          var originalText = loadedSongsMap[songName] || "";
          var resultText = transformFn(originalText, songName);

          if (typeof resultText !== "string") {
            throw new Error("Script must return a string (failed on '" + songName + "').");
          }

          if (resultText !== originalText) {
            modifiedCount++;
            updatedDrafts[songName] = resultText;
          }
        }
      } catch (execErr) {
        statusEl.style.color = "#ff5252";
        statusEl.innerText = "Execution Error: " + execErr.message;
        return;
      }

      if (isPreviewOnly) {
        statusEl.style.color = "#ffb74d";
        statusEl.innerText = "Preview: " + modifiedCount + " of " + playlistOrder.length + " song(s) will be modified.";
        return;
      }

      if (modifiedCount === 0) {
        statusEl.style.color = "#aaa";
        statusEl.innerText = "No changes detected across any loaded songs.";
        return;
      }

      // Commit transformed text into memory and flag each modified song
      Object.keys(updatedDrafts).forEach(function(songName) {
        loadedSongsMap[songName] = updatedDrafts[songName];
        modifiedSongsStatus[songName] = true;
      });

      renderPlaylist();
      saveAppStateToCache();

      // Re-render active song immediately if it was modified
      if (activeSongName && updatedDrafts[activeSongName]) {
        var rawLines = loadedSongsMap[activeSongName].split(/\r?\n/).map(function(ln) { return ln.trim(); });
        document.getElementById('prompter-stage').innerHTML = compileMarkupToHTML(rawLines);
        syncLeftChordsColumnWidth();
      }

      closeModal('batch-edit-modal');
      alert("Success! Updated " + modifiedCount + " song(s). Use 'Save All Songs' in the menu to write changes to disk.");
    }

    async function saveAllSongs() {
      if (playlistOrder.length === 0) {
        alert("No songs loaded to save.");
        return;
      }

      var modifiedSongs = playlistOrder.filter(function(songName) {
        return modifiedSongsStatus[songName] === true;
      });

      if (modifiedSongs.length === 0) {
        alert("All loaded songs are already up to date (no unsaved modifications).");
        return;
      }

      // Group modified songs by origin ("txt" vs specific ".json" filenames)
      var txtSongsToSave = [];
      var jsonOriginsWithChanges = {};

      modifiedSongs.forEach(function(songName) {
        var origin = songSourcesMap[songName] || "txt";
        if (origin === "txt") {
          txtSongsToSave.push(songName);
        } else {
          jsonOriginsWithChanges[origin] = true;
        }
      });

      var summaryMessages = [];

      // 1. Save all modified .txt songs to a chosen folder
      if (txtSongsToSave.length > 0) {
        try {
          if (!window.showDirectoryPicker) {
            alert("Your browser does not support folder batch-saving for .txt files. Please save .txt songs individually.");
          } else {
            alert("Select the folder where your " + txtSongsToSave.length + " modified .txt song(s) should be saved.");
            var dirHandle = await window.showDirectoryPicker({ mode: "readwrite" });

            for (var i = 0; i < txtSongsToSave.length; i++) {
              var sName = txtSongsToSave[i];
              var fileHandle = await dirHandle.getFileHandle(sName + ".txt", { create: true });
              var writable = await fileHandle.createWritable();
              await writable.write(loadedSongsMap[sName] || "");
              await writable.close();
              modifiedSongsStatus[sName] = false;
            }
            summaryMessages.push("Saved " + txtSongsToSave.length + " .txt file(s) to '" + dirHandle.name + "'.");
          }
        } catch (err) {
          if (err.name !== 'AbortError') {
            alert("Error saving .txt files: " + err.toString());
          }
        }
      }

      // 2. Save each modified .json setlist back to its own .json file
      var jsonFileNames = Object.keys(jsonOriginsWithChanges);
      for (var j = 0; j < jsonFileNames.length; j++) {
        var targetJsonName = jsonFileNames[j];

        // Collect all songs currently in the playlist that belong to this specific .json set
        var setOrder = [];
        var setSongsMap = {};
        playlistOrder.forEach(function(songName) {
          if (songSourcesMap[songName] === targetJsonName) {
            setOrder.push(songName);
            setSongsMap[songName] = loadedSongsMap[songName];
          }
        });

        var packageToWrite = {
          playlistOrder: setOrder,
          loadedSongsMap: setSongsMap,
          modifiedSongsStatus: {}
        };

        try {
          if (window.showSaveFilePicker) {
            alert("Select '" + targetJsonName + "' to save updates for its " + setOrder.length + " song(s).");
            var handle = await window.showSaveFilePicker({
              suggestedName: targetJsonName,
              types: [{
                description: 'JSON Setlist File',
                accept: { 'application/json': ['.json'] }
              }]
            });
            var stream = await handle.createWritable();
            await stream.write(JSON.stringify(packageToWrite, null, 2));
            await stream.close();

            // Update origin if user saved under a new name and clear modified flags for this set
            var finalName = handle.name;
            setOrder.forEach(function(songName) {
              songSourcesMap[songName] = finalName;
              modifiedSongsStatus[songName] = false;
            });
            summaryMessages.push("Updated setlist '" + finalName + "'.");
          }
        } catch (err) {
          if (err.name !== 'AbortError') {
            alert("Error saving setlist '" + targetJsonName + "': " + err.toString());
          }
        }
      }

      renderPlaylist();
      saveAppStateToCache();

      if (summaryMessages.length > 0) {
        alert("Save All Complete:\n• " + summaryMessages.join("\n• "));
      }
    }


  </script>
